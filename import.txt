// 0. Identifier constraints
// ============================================================
CREATE CONSTRAINT customer_id IF NOT EXISTS
FOR (n:Customer) REQUIRE n.customerId IS UNIQUE;
CREATE CONSTRAINT address_id IF NOT EXISTS
FOR (n:Address) REQUIRE n.addressId IS UNIQUE;
CREATE CONSTRAINT store_id IF NOT EXISTS
FOR (n:Store) REQUIRE n.storeId IS UNIQUE;
CREATE CONSTRAINT driver_id IF NOT EXISTS
FOR (n:Driver) REQUIRE n.driverId IS UNIQUE;
CREATE CONSTRAINT category_id IF NOT EXISTS
FOR (n:MenuCategory) REQUIRE n.categoryId IS UNIQUE;
CREATE CONSTRAINT menu_item_id IF NOT EXISTS
FOR (n:MenuItem) REQUIRE n.menuItemId IS UNIQUE;
CREATE CONSTRAINT variant_id IF NOT EXISTS
FOR (n:MenuVariant) REQUIRE n.variantId IS UNIQUE;
CREATE CONSTRAINT order_id IF NOT EXISTS
FOR (n:Order) REQUIRE n.orderId IS UNIQUE;
CREATE CONSTRAINT payment_id IF NOT EXISTS
FOR (n:Payment) REQUIRE n.paymentId IS UNIQUE;
CREATE CONSTRAINT delivery_id IF NOT EXISTS
FOR (n:Delivery) REQUIRE n.deliveryId IS UNIQUE;

// ============================================================
// 1. Nodes
// ============================================================
LOAD CSV WITH HEADERS FROM 'file:///customers.csv' AS row
MERGE (c:Customer {customerId: toInteger(row.CustomerID)})
SET c.firstName = row.FirstName,
    c.lastName = row.LastName,
    c.email = row.Email,
    c.phone = row.Phone,
    c.loyaltyPoints = toInteger(row.LoyaltyPoints);

LOAD CSV WITH HEADERS FROM 'file:///addresses.csv' AS row
MERGE (a:Address {addressId: toInteger(row.AddressID)})
SET a.streetAddress = row.StreetAddress,
    a.suburb = row.Suburb,
    a.state = row.State,
    a.postcode = row.Postcode;

LOAD CSV WITH HEADERS FROM 'file:///stores.csv' AS row
MERGE (s:Store {storeId: toInteger(row.StoreID)})
SET s.storeName = row.StoreName,
    s.streetAddress = row.StreetAddress,
    s.suburb = row.Suburb,
    s.state = row.State,
    s.postcode = row.Postcode,
    s.phone = row.Phone;

LOAD CSV WITH HEADERS FROM 'file:///drivers.csv' AS row
MERGE (d:Driver {driverId: toInteger(row.DriverID)})
SET d.driverId = toInteger(row.DriverID),
    d.firstName = row.FirstName,
    d.lastName = row.LastName,
    d.phone = row.Phone,
    d.driverStatus = row.DriverStatus;

LOAD CSV WITH HEADERS FROM 'file:///categories.csv' AS row
MERGE (c:MenuCategory {categoryId: toInteger(row.CategoryID)})
SET c.categoryName = row.CategoryName,
    c.description = row.Description;

LOAD CSV WITH HEADERS FROM 'file:///menu_items.csv' AS row
MERGE (m:MenuItem {menuItemId: toInteger(row.MenuItemID)})
SET m.itemName = row.ItemName,
    m.description = row.Description,
    m.availabilityStatus = row.AvailabilityStatus;

LOAD CSV WITH HEADERS FROM 'file:///menu_variants.csv' AS row
MERGE (v:MenuVariant {variantId: toInteger(row.VariantID)})
SET v.menuItemId = toInteger(row.MenuItemID),
    v.size = row.Size,
    v.crustType = CASE WHEN row.CrustType = '' THEN null ELSE row.CrustType END,
    v.currentPrice = toFloat(row.CurrentPrice),
    v.availabilityStatus = row.AvailabilityStatus;

LOAD CSV WITH HEADERS FROM 'file:///orders.csv' AS row
MERGE (o:Order {orderId: toInteger(row.OrderID)})
SET o.orderDateTime = datetime(row.OrderDateTime),
    o.orderStatus = row.OrderStatus,
    o.fulfilmentType = row.FulfilmentType
WITH o, row
FOREACH (_ IN CASE WHEN row.FulfilmentType = 'Pickup' THEN [1] ELSE [] END |
    SET o:PickupOrder)
FOREACH (_ IN CASE WHEN row.FulfilmentType = 'Delivery' THEN [1] ELSE [] END |
    SET o:DeliveryOrder);

LOAD CSV WITH HEADERS FROM 'file:///payments.csv' AS row
MERGE (p:Payment {paymentId: toInteger(row.PaymentID)})
SET p.paymentMethod = row.PaymentMethod,
    p.paymentAmount = toFloat(row.PaymentAmount),
    p.paymentStatus = row.PaymentStatus,
    p.paymentDateTime = datetime(row.PaymentDateTime),
    p.voucherCode = CASE WHEN row.VoucherCode = '' THEN null ELSE row.VoucherCode END,
    p.priorityFee = toFloat(row.PriorityFee);

LOAD CSV WITH HEADERS FROM 'file:///deliveries.csv' AS row
MERGE (d:Delivery {deliveryId: toInteger(row.DeliveryID)})
SET d.trackerStatus = row.TrackerStatus,
    d.dispatchTime = datetime(row.DispatchTime),
    d.deliveredTime = CASE WHEN row.DeliveredTime = '' THEN null ELSE datetime(row.DeliveredTime) END,
    d.estimatedDeliveryTime = datetime(row.EstimatedDeliveryTime);

// ============================================================
// 2. Relationships
// ============================================================
LOAD CSV WITH HEADERS FROM 'file:///addresses.csv' AS row
MATCH (c:Customer {customerId: toInteger(row.CustomerID)}),
      (a:Address {addressId: toInteger(row.AddressID)})
MERGE (c)-[:HAS_ADDRESS]->(a);

LOAD CSV WITH HEADERS FROM 'file:///orders.csv' AS row
MATCH (c:Customer {customerId: toInteger(row.CustomerID)}),
      (o:Order {orderId: toInteger(row.OrderID)}),
      (s:Store {storeId: toInteger(row.StoreID)})
MERGE (c)-[:PLACES]->(o)
MERGE (s)-[:PROCESSES]->(o);

LOAD CSV WITH HEADERS FROM 'file:///drivers.csv' AS row
MATCH (s:Store {storeId: toInteger(row.StoreID)}),
      (d:Driver {driverId: toInteger(row.DriverID)})
MERGE (s)-[:ASSOCIATED_WITH]->(d);

LOAD CSV WITH HEADERS FROM 'file:///menu_items.csv' AS row
MATCH (c:MenuCategory {categoryId: toInteger(row.CategoryID)}),
      (m:MenuItem {menuItemId: toInteger(row.MenuItemID)})
MERGE (c)-[:CLASSIFIES]->(m);

LOAD CSV WITH HEADERS FROM 'file:///menu_variants.csv' AS row
MATCH (m:MenuItem {menuItemId: toInteger(row.MenuItemID)}),
      (v:MenuVariant {variantId: toInteger(row.VariantID)})
MERGE (m)-[:HAS_VARIANT]->(v);

// ORDER_ITEM -> CONTAINS relationship properties.
// The Assignment 2 composite key (OrderID, LineNumber) is checked after import.
LOAD CSV WITH HEADERS FROM 'file:///order_items.csv' AS row
MATCH (o:Order {orderId: toInteger(row.OrderID)}),
      (v:MenuVariant {variantId: toInteger(row.VariantID)})
CREATE (o)-[:CONTAINS {
    lineNumber: toInteger(row.LineNumber),
    quantity: toInteger(row.Quantity),
    unitPrice: toFloat(row.UnitPrice),
    specialInstructions: CASE WHEN row.SpecialInstructions = '' THEN null ELSE row.SpecialInstructions END
}]->(v);

LOAD CSV WITH HEADERS FROM 'file:///payments.csv' AS row
MATCH (o:Order {orderId: toInteger(row.OrderID)}),
      (p:Payment {paymentId: toInteger(row.PaymentID)})
MERGE (o)-[:SETTLED_BY]->(p);

LOAD CSV WITH HEADERS FROM 'file:///deliveries.csv' AS row
MATCH (o:Order:DeliveryOrder {orderId: toInteger(row.OrderID)}),
      (d:Delivery {deliveryId: toInteger(row.DeliveryID)}),
      (dr:Driver {driverId: toInteger(row.DriverID)}),
      (a:Address {addressId: toInteger(row.AddressID)})
MERGE (o)-[:FULFILLED_BY]->(d)
MERGE (dr)-[:PERFORMS]->(d)
MERGE (a)-[:DESTINATION_FOR]->(d);

// ============================================================
// 3. Operational indexes
// ============================================================
CREATE RANGE INDEX menu_item_name IF NOT EXISTS
FOR (m:MenuItem) ON (m.itemName);
CREATE RANGE INDEX order_status IF NOT EXISTS
FOR (o:Order) ON (o.orderStatus);
CREATE RANGE INDEX payment_status IF NOT EXISTS
FOR (p:Payment) ON (p.paymentStatus);
CREATE RANGE INDEX delivery_status IF NOT EXISTS
FOR (d:Delivery) ON (d.trackerStatus);

// ============================================================
// 4. Derived PURCHASED layer for Task 4 analytics
// ============================================================
MATCH (c:Customer)-[:PLACES]->(o:Order)-[line:CONTAINS]->(v:MenuVariant)
WITH c, v, sum(line.quantity) AS quantity
MERGE (c)-[p:PURCHASED]->(v)
SET p.quantity = quantity;

// ============================================================
// 5. Affinity features for Task 5 similarity
//    Category + order status, derived only from the operational graph.
// ============================================================
MATCH (c:Customer)-[:PLACES]->(:Order)-[:CONTAINS]->(:MenuVariant)
      <-[:HAS_VARIANT]-(:MenuItem)-[:CLASSIFIES]->(cat:MenuCategory)
WITH DISTINCT c, 'CATEGORY:' + toString(cat.categoryId) AS key,
     cat.categoryName AS value
MERGE (f:AffinityFeature {featureKey: key})
SET f.featureType = 'Category',
    f.featureValue = value
MERGE (c)-[:HAS_AFFINITY_FEATURE]->(f);

MATCH (c:Customer)-[:PLACES]->(o:Order)
WITH DISTINCT c, 'STATUS:' + o.orderStatus AS key,
     o.orderStatus AS value
MERGE (f:AffinityFeature {featureKey: key})
SET f.featureType = 'OrderStatus',
    f.featureValue = value
MERGE (c)-[:HAS_AFFINITY_FEATURE]->(f);

// ============================================================
// 6. 14D Customer behaviour vector for KNN
//    10 category-spend values + 4 order-status counts.
// ============================================================
MATCH (c:Customer)
CALL {
    WITH c
    OPTIONAL MATCH (c)-[:PLACES]->(:Order)-[line:CONTAINS]->(:MenuVariant)
          <-[:HAS_VARIANT]-(:MenuItem)-[:CLASSIFIES]->(cat:MenuCategory)
    RETURN
      sum(CASE WHEN cat.categoryId = 1 THEN line.quantity * line.unitPrice ELSE 0 END) AS cat1,
      sum(CASE WHEN cat.categoryId = 2 THEN line.quantity * line.unitPrice ELSE 0 END) AS cat2,
      sum(CASE WHEN cat.categoryId = 3 THEN line.quantity * line.unitPrice ELSE 0 END) AS cat3,
      sum(CASE WHEN cat.categoryId = 4 THEN line.quantity * line.unitPrice ELSE 0 END) AS cat4,
      sum(CASE WHEN cat.categoryId = 5 THEN line.quantity * line.unitPrice ELSE 0 END) AS cat5,
      sum(CASE WHEN cat.categoryId = 6 THEN line.quantity * line.unitPrice ELSE 0 END) AS cat6,
      sum(CASE WHEN cat.categoryId = 7 THEN line.quantity * line.unitPrice ELSE 0 END) AS cat7,
      sum(CASE WHEN cat.categoryId = 8 THEN line.quantity * line.unitPrice ELSE 0 END) AS cat8,
      sum(CASE WHEN cat.categoryId = 9 THEN line.quantity * line.unitPrice ELSE 0 END) AS cat9,
      sum(CASE WHEN cat.categoryId = 10 THEN line.quantity * line.unitPrice ELSE 0 END) AS cat10
}
CALL {
    WITH c
    OPTIONAL MATCH (c)-[:PLACES]->(o:Order)
    RETURN
      count(CASE WHEN o.orderStatus = 'Completed' THEN 1 END) AS completed,
      count(CASE WHEN o.orderStatus = 'Preparing' THEN 1 END) AS preparing,
      count(CASE WHEN o.orderStatus = 'Ready for Pickup' THEN 1 END) AS readyForPickup,
      count(CASE WHEN o.orderStatus = 'Out for Delivery' THEN 1 END) AS outForDelivery
}
SET c.behaviourVector = [
    toFloat(coalesce(cat1,0)), toFloat(coalesce(cat2,0)),
    toFloat(coalesce(cat3,0)), toFloat(coalesce(cat4,0)),
    toFloat(coalesce(cat5,0)), toFloat(coalesce(cat6,0)),
    toFloat(coalesce(cat7,0)), toFloat(coalesce(cat8,0)),
    toFloat(coalesce(cat9,0)), toFloat(coalesce(cat10,0)),
    toFloat(completed), toFloat(preparing),
    toFloat(readyForPickup), toFloat(outForDelivery)
];
